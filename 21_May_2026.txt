use vitcampus;

select * from workers;

select fname, dept, sal from workers where dept='IT' order by sal;

select count(work_id) from workers where dept='Marketing';

select dept, count(dept) as number from workers group by dept order by count(dept) asc;

create table winners(
	num int
);

insert into winners values (1), (3), (22), (6);

select * from workers limit 2 offset 3;

select * from workers where work_id in (select num from winners);

select dept, sum(sal) from workers group by dept order by sum(sal) asc limit 1;

select * from workers where sal=(select distinct sal from workers as sub where sub.dept=workers.dept order by sal asc limit 1 offset 2);
